#!/bin/bash
set -e

# GET THE NPM-HOSTED LAST REPO URL
last_npm_hosted_URL=$(curl -s -u team3:team3 'http://144.126.201.150:8081/service/rest/v1/components?repository=npm-hosted&sort=version' \
        | jq -r '.items[0].assets[0].downloadUrl')
printf "\nLatest NPM HOSTED URL: $last_npm_hosted_URL\n"

# EXECUTE A COMMAND TO FETCH THE LATEST ARTIFACT WITH THE DOWNLOAD URL
artifact_name=$(basename "$last_npm_hosted_URL")
printf "\nDownloading $last_npm_hosted_URL ...\n"
curl -s -u team3:team3 -X GET "$last_npm_hosted_URL" --output "$artifact_name"
printf "$artifact_name Downloaded\n"

# GET FILE TO EXECUTE
printf "\nIdentification of package name and file to execute\n"
package_name=$(tar -xOzf "$artifact_name" package/package.json | jq -r '.name')
file_to_execute=$(tar -xOzf "$artifact_name" package/package.json | jq -r '.main')
printf "Package name: $package_name\n"
printf "Main file to execute: $file_to_execute\n"

# CLEAN OLD INSTALLATION
printf "\nClean old installation\n"
rm -rf node_modules package.json package-lock.json
npm init -y >/dev/null 2>&1
printf "Done\n"

# INSTALLATION OF NODE MODULES
printf "\nInstallation of needed modules\n"
npm install "./$artifact_name"
printf "node_modules installed\n"

# RUN IT USING NODE  
printf "\nStarting application ...\n"
node "node_modules/$package_name/$file_to_execute"
